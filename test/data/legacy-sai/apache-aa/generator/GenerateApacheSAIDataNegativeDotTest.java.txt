package org.apache.cassandra.index.sai;

import java.nio.file.Files;
import java.nio.file.Path;
import java.nio.file.Paths;
import java.nio.file.StandardCopyOption;
import java.util.ArrayList;
import java.util.List;
import java.util.Random;
import java.util.stream.Stream;

import org.junit.Test;

import org.apache.cassandra.io.sstable.format.SSTableFormat;
import org.apache.cassandra.io.sstable.format.SSTableReader;

public class GenerateApacheSAIDataNegativeDotTest extends SAITester
{
    @Test
    public void generate() throws Throwable
    {
        createTable("CREATE TABLE %s (pk int PRIMARY KEY, st set<text>, v int)");
        disableCompaction();
        createIndex("CREATE INDEX negative_st ON %s(st) USING 'sai'");
        execute("INSERT INTO %s (pk, st, v) VALUES (1, {'x'}, 1) USING TIMESTAMP 1000");
        execute("INSERT INTO %s (pk, st, v) VALUES (2, {'y'}, 1) USING TIMESTAMP 1000");
        flush();
        execute("DELETE st FROM %s USING TIMESTAMP 2000 WHERE pk = 1");
        execute("INSERT INTO %s (pk, st, v) VALUES (3, {'z'}, 1) USING TIMESTAMP 1000");
        flush();
        copy("negative");

        createTable("CREATE TABLE %s (pk int PRIMARY KEY, v vector<float, 8>)");
        disableCompaction();
        createIndex("CREATE INDEX dot_v ON %s(v) USING 'sai' WITH OPTIONS = {'similarity_function': 'DOT_PRODUCT'}");
        Random rnd = new Random(12345);
        for (int i = 0; i < 4000; i++)
        {
            List<Float> v = new ArrayList<>();
            float magnitude = i < 200 ? 10f : 1f;
            for (int d = 0; d < 8; d++)
                v.add((float) rnd.nextGaussian() * magnitude);
            execute("INSERT INTO %s (pk, v) VALUES (?, ?)", i, vector(v.toArray(new Float[0])));
        }
        flush();
        copy("dot");
    }

    void copy(String name) throws Exception
    {
        Path out = Paths.get("/tmp/apache-sai-gen/data", name);
        Files.createDirectories(out);
        for (SSTableReader s : getCurrentColumnFamilyStore().getLiveSSTables())
        {
            Path data = s.descriptor.fileFor(SSTableFormat.Components.DATA).toPath();
            String prefix = data.getFileName().toString().replace("Data.db", "");
            try (Stream<Path> files = Files.list(data.getParent()))
            {
                for (Path src : (Iterable<Path>) files.filter(f -> f.getFileName().toString().startsWith(prefix))::iterator)
                    Files.copy(src, out.resolve(src.getFileName()), StandardCopyOption.REPLACE_EXISTING);
            }
        }
        System.out.println("GENERATED " + out);
    }
}
