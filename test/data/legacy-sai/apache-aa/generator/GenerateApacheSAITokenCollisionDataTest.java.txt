package org.apache.cassandra.index.sai;

import java.nio.file.Files;
import java.nio.file.Path;
import java.nio.file.Paths;
import java.nio.file.StandardCopyOption;
import java.util.ArrayList;
import java.util.Comparator;
import java.util.List;
import java.util.function.BiConsumer;
import java.util.stream.Stream;

import org.junit.Test;

import org.apache.cassandra.Util;
import org.apache.cassandra.db.DecoratedKey;
import org.apache.cassandra.dht.IPartitioner;
import org.apache.cassandra.dht.Token;
import org.apache.cassandra.io.sstable.format.SSTableFormat;
import org.apache.cassandra.io.sstable.format.SSTableReader;
import org.apache.cassandra.utils.ByteBufferUtil;

/**
 * Writes SSTables whose distinct partition keys share tokens (three groups of six keys with one token each, and keys
 * of other tokens between the groups), with SAI indexes, for DataStax's TokenCollisionQueryTest. Each schema is written
 * spread over six SSTables, each with one key of each group ("-spread"), then major compacted into one SSTable. The
 * keys and the values are those of TokenCollisionQueryTest: run with -Dsai.out=<dir>.
 */
public class GenerateApacheSAITokenCollisionDataTest extends SAITester
{
    private static final int KEYS_PER_TOKEN = 6;

    private List<DecoratedKey> keys;
    private List<DecoratedKey> collidingKeys;

    @Test
    public void generateSkinny() throws Throwable
    {
        createTable("CREATE TABLE %s (pk blob PRIMARY KEY, a int, b int)");
        createIndex("CREATE INDEX collisions_skinny_a ON %s(a) USING 'sai'");
        createIndex("CREATE INDEX collisions_skinny_b ON %s(b) USING 'sai'");
        generate("collisions-skinny", (i, key) -> execute("INSERT INTO %s (pk, a, b) VALUES (?, ?, ?)", key.getKey(), i % 3, i % 2));
    }

    @Test
    public void generateWide() throws Throwable
    {
        createTable("CREATE TABLE %s (pk blob, ck int, s int static, a int, b int, PRIMARY KEY (pk, ck))");
        createIndex("CREATE INDEX collisions_wide_s ON %s(s) USING 'sai'");
        createIndex("CREATE INDEX collisions_wide_a ON %s(a) USING 'sai'");
        createIndex("CREATE INDEX collisions_wide_b ON %s(b) USING 'sai'");
        generate("collisions-wide", (i, key) -> {
            execute("INSERT INTO %s (pk, s) VALUES (?, ?)", key.getKey(), i % 2);
            for (int ck = 0; ck < 3; ck++)
                execute("INSERT INTO %s (pk, ck, a, b) VALUES (?, ?, ?, ?)", key.getKey(), ck, (i + ck) % 3, (i * ck) % 2);
        });
    }

    private void generate(String name, BiConsumer<Integer, DecoratedKey> insertPartition) throws Throwable
    {
        disableCompaction();
        keys = keys();
        for (int sstable = 0; sstable < KEYS_PER_TOKEN; sstable++)
        {
            for (int i = 0; i < keys.size(); i++)
                if (sstableOf(keys.get(i)) == sstable)
                    insertPartition.accept(i, keys.get(i));
            flush();
        }
        copy(name + "-spread");
        compact();
        copy(name);
        System.out.println("GENERATED " + name + " keys=" + keys.size());
    }

    private void copy(String name) throws Exception
    {
        Path out = Paths.get(System.getProperty("sai.out", "/tmp/apache-sai-collisions"), name);
        Files.createDirectories(out);
        for (SSTableReader s : getCurrentColumnFamilyStore().getLiveSSTables())
        {
            Path data = s.descriptor.fileFor(SSTableFormat.Components.DATA).toPath();
            String prefix = data.getFileName().toString().replace("Data.db", "");
            try (Stream<Path> files = Files.list(data.getParent()))
            {
                for (Path src : (Iterable<Path>) files.filter(f -> f.getFileName().toString().startsWith(prefix))::iterator)
                    Files.copy(src, out.resolve(src.getFileName()), StandardCopyOption.REPLACE_EXISTING);
            }
        }
    }

    // as TokenCollisionQueryTest#keys
    private List<DecoratedKey> keys()
    {
        IPartitioner partitioner = getCurrentColumnFamilyStore().getPartitioner();
        List<DecoratedKey> keys = new ArrayList<>();
        collidingKeys = new ArrayList<>();
        List<Token> tokens = new ArrayList<>();
        for (String prefix : new String[]{ "group0", "group1", "group2" })
        {
            for (int i = 0; i < KEYS_PER_TOKEN; i++)
                collidingKeys.add(partitioner.decorateKey(Util.generateMurmurCollision(ByteBufferUtil.bytes(prefix), (byte) i)));
            tokens.add(collidingKeys.get(collidingKeys.size() - 1).getToken());
        }
        collidingKeys.sort(Comparator.naturalOrder());
        tokens.sort(Comparator.naturalOrder());
        keys.addAll(collidingKeys);
        int beforeMiddle = 0;
        int afterMiddle = 0;
        for (int i = 0; beforeMiddle < KEYS_PER_TOKEN || afterMiddle < KEYS_PER_TOKEN; i++)
        {
            DecoratedKey key = partitioner.decorateKey(ByteBufferUtil.bytes(i));
            if (key.getToken().compareTo(tokens.get(0)) > 0 && key.getToken().compareTo(tokens.get(1)) < 0 && beforeMiddle < KEYS_PER_TOKEN)
            {
                beforeMiddle++;
                keys.add(key);
            }
            else if (key.getToken().compareTo(tokens.get(1)) > 0 && key.getToken().compareTo(tokens.get(2)) < 0 && afterMiddle < KEYS_PER_TOKEN)
            {
                afterMiddle++;
                keys.add(key);
            }
        }
        keys.sort(Comparator.naturalOrder());
        return keys;
    }

    // as TokenCollisionQueryTest#sstableOf
    private int sstableOf(DecoratedKey key)
    {
        int colliding = collidingKeys.indexOf(key);
        if (colliding >= 0)
            return colliding % KEYS_PER_TOKEN;
        int i = keys.indexOf(key);
        int previousColliding = i;
        while (previousColliding >= 0 && !collidingKeys.contains(keys.get(previousColliding)))
            previousColliding--;
        return (i - previousColliding - 1) % KEYS_PER_TOKEN;
    }
}
