package org.apache.cassandra.index.sai;

import java.io.File;
import java.nio.file.Files;
import java.nio.file.Path;
import java.nio.file.StandardCopyOption;
import java.util.stream.Stream;

import org.junit.Test;

import org.apache.cassandra.db.ColumnFamilyStore;
import org.apache.cassandra.io.sstable.Component;
import org.apache.cassandra.io.sstable.format.SSTableFormat;
import org.apache.cassandra.io.sstable.format.SSTableReader;

/**
 * Generator of test data: writes Apache 5.0 SAI indexes for the DataStax compatibility tests.
 * Output: -Dsai.out=<dir>, one sub-directory per table.
 */
public class GenerateApacheSAIDataTest extends SAITester
{
    @Test
    public void generate() throws Throwable
    {
        String out = System.getProperty("sai.out", "/tmp/apache-sai-data");
        String ks = "apache_sai";
        schemaChange("CREATE KEYSPACE IF NOT EXISTS " + ks + " WITH replication = {'class': 'SimpleStrategy', 'replication_factor': 1}");

        // skinny: literal, numeric, vector
        schemaChange("CREATE TABLE " + ks + ".skinny (pk int PRIMARY KEY, s text, n int, v vector<float, 3>)");
        schemaChange("CREATE INDEX skinny_s ON " + ks + ".skinny(s) USING 'sai'");
        schemaChange("CREATE INDEX skinny_n ON " + ks + ".skinny(n) USING 'sai'");
        schemaChange("CREATE INDEX skinny_v ON " + ks + ".skinny(v) USING 'sai'");
        // wide: literal, numeric, plus a static column
        schemaChange("CREATE TABLE " + ks + ".wide (pk int, ck int, st int static, s text, n bigint, PRIMARY KEY (pk, ck))");
        schemaChange("CREATE INDEX wide_s ON " + ks + ".wide(s) USING 'sai'");
        schemaChange("CREATE INDEX wide_n ON " + ks + ".wide(n) USING 'sai'");
        schemaChange("CREATE INDEX wide_st ON " + ks + ".wide(st) USING 'sai'");

        for (int sstable = 0; sstable < 2; sstable++)
        {
            for (int i = sstable * 50; i < (sstable + 1) * 50; i++)
            {
                execute("INSERT INTO " + ks + ".skinny (pk, s, n, v) VALUES (?, ?, ?, ?)",
                        i, "s" + (i % 10), i, vector((float) i, (float) (i % 7), 1.0f));
                for (int ck = 0; ck < 4; ck++)
                    execute("INSERT INTO " + ks + ".wide (pk, ck, st, s, n) VALUES (?, ?, ?, ?, ?)",
                            i, ck, i % 5, "w" + (ck % 3), (long) (i * 10 + ck));
            }
            flush(ks, "skinny");
            flush(ks, "wide");
        }

        for (String table : new String[]{ "skinny", "wide" })
        {
            ColumnFamilyStore cfs = ColumnFamilyStore.getIfExists(ks, table);
            Path dir = new File(out, table).toPath();
            Files.createDirectories(dir);
            for (SSTableReader sstable : cfs.getLiveSSTables())
            {
                Path data = sstable.descriptor.fileFor(SSTableFormat.Components.DATA).toPath();
                String prefix = data.getFileName().toString().replace("Data.db", "");
                try (Stream<Path> files = Files.list(data.getParent()))
                {
                    for (Path src : (Iterable<Path>) files.filter(f -> f.getFileName().toString().startsWith(prefix))::iterator)
                        Files.copy(src, dir.resolve(src.getFileName()), StandardCopyOption.REPLACE_EXISTING);
                }
            }
            System.out.println("SAIGEN " + table + " -> " + dir + " : " + cfs.getLiveSSTables().size() + " sstables");
        }
    }
}
