package org.apache.cassandra.index.sai;

import java.nio.file.Files;
import java.nio.file.Path;
import java.nio.file.Paths;
import java.nio.file.StandardCopyOption;
import java.util.ArrayList;
import java.util.Comparator;
import java.util.List;
import java.util.stream.Stream;

import org.junit.Test;

import org.apache.cassandra.db.marshal.Int32Type;
import org.apache.cassandra.dht.Murmur3Partitioner;
import org.apache.cassandra.io.sstable.format.SSTableFormat;
import org.apache.cassandra.io.sstable.format.SSTableReader;

public class GenerateApacheSAIDataMismatchTest extends SAITester
{
    @Test
    public void generate() throws Throwable
    {
        createTable("CREATE TABLE %s (pk int PRIMARY KEY, s text, v int, x text)");
        disableCompaction();
        createIndex("CREATE INDEX mismatch_s ON %s(s) USING 'sai'");
        List<Integer> keys = new ArrayList<>();
        for (int i = 0; i < 10; i++)
            keys.add(i);
        keys.sort(Comparator.comparing(i -> Murmur3Partitioner.instance.decorateKey(Int32Type.instance.decompose(i))));
        execute("DELETE s FROM %s USING TIMESTAMP 2000 WHERE pk=?", keys.get(0));
        for (int n = 1; n < keys.size(); n++)
        {
            if (n == 1 || n == 2)
                execute("INSERT INTO %s (pk,s,v) VALUES (?,'match',1) USING TIMESTAMP 1000", keys.get(n));
            else
                execute("INSERT INTO %s (pk,v) VALUES (?,1) USING TIMESTAMP 1000", keys.get(n));
        }
        flush();

        Path out = Paths.get("/tmp/apache-sai-gen/data/mismatch");
        Files.createDirectories(out);
        for (SSTableReader s : getCurrentColumnFamilyStore().getLiveSSTables())
        {
            Path data = s.descriptor.fileFor(SSTableFormat.Components.DATA).toPath();
            String prefix = data.getFileName().toString().replace("Data.db", "");
            try (Stream<Path> files = Files.list(data.getParent()))
            {
                for (Path src : (Iterable<Path>) files.filter(f -> f.getFileName().toString().startsWith(prefix))::iterator)
                    Files.copy(src, out.resolve(src.getFileName()), StandardCopyOption.REPLACE_EXISTING);
            }
        }
        System.out.println("GENERATED MISMATCH keys=" + keys + " expected=" + keys.subList(1, 3));
    }
}
